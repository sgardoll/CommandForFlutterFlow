/**
 * Tests error handling for various HTTP status codes.
 * @returns {Promise<Object>} Test results
 */
async function testErrorHandling() {
  console.log('🧪 Testing error handling...');
  
  const testCases = [
    { code: 401, name: 'Authentication error' },
    { code: 403, name: 'Forbidden error' },
    { code: 404, name: 'Not found error' },
    { code: 422, name: 'Validation error' },
    { code: 429, name: 'Rate limit error' },
    { code: 500, name: 'Server error' },
  ];
  
  const results = [];
  
  for (const testCase of testCases) {
    console.log(`  Testing ${testCase.name}...`);
    
    mockFlutterFlowApi({ 
      shouldSucceed: false, 
      delay: 100,
      response: { message: 'Test error' }
    });
    
    // Override the mock to return specific status
    window.fetch = async (url, config) => {
      await sleep(100);
      return {
        ok: false,
        status: testCase.code,
        json: async () => ({ message: 'Test error message' }),
        clone: function() { return this; },
        text: async () => 'Test error',
      };
    };
    
    try {
      await saveApiKey('flutterflow', 'test_key_12345678901234567890');
      await saveApiKey('flutterflow_project_id', 'test-project');
      
      const result = await executeCommit('test code', {
        artifactType: 'CustomWidget',
        artifactName: 'Test',
      });
      
      const categorized = categorizeError(new Error('Test'), testCase.code);
      
      results.push({
        name: testCase.name,
        code: testCase.code,
        handled: !result.success,
        errorCategory: categorized.category,
        pass: !result.success && categorized.category !== 'UNKNOWN',
      });
    } catch (error) {
      results.push({
        name: testCase.name,
        code: testCase.code,
        handled: true,
        error: error.message,
        pass: true,
      });
    }
  }
  
  unmockFlutterFlowApi();
  
  const allPassed = results.every(r => r.pass);
  
  console.log('  Results:');
  results.forEach(r => {
    console.log(`    ${r.pass ? '✅' : '❌'} ${r.name} (${r.code}) - ${r.errorCategory || 'N/A'}`);
  });
  
  return {
    success: allPassed,
    results,
    message: allPassed ? 'All error handling tests passed!' : 'Some tests failed',
  };
}

/**
 * Tests edge cases in code validation.
 * @returns {Object} Test results
 */
function testCodeValidationEdgeCases() {
  console.log('🧪 Testing code validation edge cases...');
  
  const testCases = [
    {
      name: 'Empty code',
      code: '',
      shouldPass: false,
    },
    {
      name: 'Code with main()',
      code: 'void main() { runApp(MyApp()); }',
      shouldPass: false,
    },
    {
      name: 'Code with imports',
      code: "import 'package:flutter/material.dart';\nclass Test {}",
      shouldPass: false,
    },
    {
      name: 'Very large code',
      code: 'x'.repeat(200000),
      shouldPass: false,
    },
    {
      name: 'Valid widget code',
      code: 'class TestWidget extends StatelessWidget { @override Widget build(BuildContext context) { return Container(); } }',
      shouldPass: true,
    },
    {
      name: 'Code with null safety issues',
      code: 'String? name; print(name!);',
      shouldPass: true, // Warning but not error
    },
  ];
  
  const results = testCases.map(tc => {
    const result = validateDartFile('test.dart', tc.code);
    const passed = result.valid === tc.shouldPass;
    
    return {
      name: tc.name,
      expected: tc.shouldPass ? 'pass' : 'fail',
      actual: result.valid ? 'pass' : 'fail',
      errors: result.errors,
      pass: passed,
    };
  });
  
  const allPassed = results.every(r => r.pass);
  
  console.log('  Results:');
  results.forEach(r => {
    console.log(`    ${r.pass ? '✅' : '❌'} ${r.name}: expected ${r.expected}, got ${r.actual}`);
    if (!r.pass && r.errors.length > 0) {
      console.log(`       Errors: ${r.errors.join(', ')}`);
    }
  });
  
  return {
    success: allPassed,
    results,
    message: allPassed ? 'All validation tests passed!' : 'Some tests failed',
  };
}

      id: 'ab',
      shouldPass: false,
    },
    {
      name: 'Project ID with spaces',
      id: 'my project 123',
      shouldPass: false,
    },
  ];
  
  const results = testCases.map(tc => {

      passed = validateFlutterFlowApiKey(tc.key) === tc.shouldPass;

    return {
      name: tc.name,

      expected: tc.shouldPass ? 'valid' : 'invalid',
      pass: passed,
    };
  });
  
  const allPassed = results.every(r => r.pass);
  
  console.log('  Results:');
  results.forEach(r => {

  });
  
  return {
    success: allPassed,
    results,
    message: allPassed ? 'All credential tests passed!' : 'Some tests failed',
  };
}

/**
 * Tests state transitions during commit.
 * @returns {Promise<Object>} Test results
 */
async function testStateTransitions() {
  console.log('🧪 Testing state transitions...');
  
  mockFlutterFlowApi({ shouldSucceed: true, delay: 100 });
  
  const states = [];
  
  // Listen for state changes
  const stateListener = (e) => {
    states.push(e.detail.state);
  };
  window.addEventListener('commitStateChange', stateListener);
  
  await saveApiKey('flutterflow', 'test_key_12345678901234567890');
  await saveApiKey('flutterflow_project_id', 'test-project');
  
  await executeCommit('class Test {}', {
    artifactType: 'CustomWidget',
    artifactName: 'Test',
  });
  
  window.removeEventListener('commitStateChange', stateListener);
  unmockFlutterFlowApi();
  
  const expectedStates = [
    CommitState.PREPARING,
    CommitState.VALIDATING,
    CommitState.PUSHING,
    CommitState.SUCCESS,
  ];
  
  const allPresent = expectedStates.every(s => states.includes(s));
  const inOrder = JSON.stringify(states.filter(s => expectedStates.includes(s))) === 
                  JSON.stringify(expectedStates);
  
  console.log('  States visited:', states.join(' → '));
  console.log(`    ${allPresent ? '✅' : '❌'} All expected states present`);
  console.log(`    ${inOrder ? '✅' : '❌'} States in correct order`);
  
  return {
    success: allPresent && inOrder,
    states,
    expectedStates,
    message: (allPresent && inOrder) ? 'State transitions correct!' : 'State transition issues found',
  };
}
