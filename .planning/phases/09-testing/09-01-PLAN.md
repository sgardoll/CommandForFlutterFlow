/**
 * Test utilities for FlutterFlow commit functionality.
 * These functions help verify the commit flow without making real API calls.
 */

/**
 * Mocks the FlutterFlow API for testing.
 * @param {Object} options - Mock options
 * @param {boolean} options.shouldSucceed - Whether the mock should succeed
 * @param {number} [options.delay=1000] - Simulated network delay
 * @param {Object} [options.response] - Custom response data
 */
function mockFlutterFlowApi(options = {}) {
  const { shouldSucceed = true, delay = 1000, response = {} } = options;
  
  // Store original fetch
  window._originalFetch = window.fetch;
  
  // Mock fetch
  window.fetch = async (url, config) => {
    console.log('[MOCK] Fetch called:', url);
    
    // Simulate network delay
    await sleep(delay);
    
    if (shouldSucceed) {
      return {
        ok: true,
        status: 200,
        json: async () => ({
          value: JSON.stringify({}),
          ...response,
        }),
        clone: function() { return this; },
        text: async () => JSON.stringify(response),
      };
    } else {
      return {
        ok: false,
        status: 401,
        json: async () => ({ message: 'Unauthorized' }),
        clone: function() { return this; },
        text: async () => 'Unauthorized',
      };
    }
  };
}

/**
 * Restores the original fetch function.
 */
function unmockFlutterFlowApi() {
  if (window._originalFetch) {
    window.fetch = window._originalFetch;
    delete window._originalFetch;
  }
}

/**
 * Runs a test commit with mock API.
 * @returns {Promise<Object>} Test result
 */
async function testCommitFlow() {
  console.log('🧪 Testing commit flow with mock API...');
  
  // Mock the API
  mockFlutterFlowApi({ shouldSucceed: true, delay: 500 });
  
  try {
    // Set test credentials
    await saveApiKey('flutterflow', 'test_api_key_12345678901234567890');
    await saveApiKey('flutterflow_project_id', 'test-project-123');
    
    // Test code
    const testCode = `
class TestWidget extends StatefulWidget {
  @override
  _TestWidgetState createState() => _TestWidgetState();
}

class _TestWidgetState extends State<TestWidget> {
  @override
  Widget build(BuildContext context) {
    return Container(
      child: Text('Test'),
    );
  }
}
    `.trim();
    
    // Execute commit
    const result = await executeCommit(testCode, {
      artifactType: 'CustomWidget',
      artifactName: 'TestWidget',
    });
    
    console.log('✅ Test result:', result);
    
    // Verify result structure
    const checks = [
      { name: 'Result exists', pass: !!result },
      { name: 'Success flag', pass: typeof result.success === 'boolean' },
      { name: 'Has message or error', pass: !!(result.message || result.error) },
      { name: 'State tracking', pass: commitState.currentState === CommitState.SUCCESS },
    ];
    
    const allPassed = checks.every(c => c.pass);
    
    return {
      success: allPassed && result.success,
      checks,
      result,
      message: allPassed ? 'All tests passed!' : 'Some tests failed',
    };
    
  } catch (error) {
    console.error('❌ Test failed:', error);
    return {
      success: false,
      error: error.message,
      checks: [],
    };
  } finally {
    unmockFlutterFlowApi();
  }
}
