/**
 * Attempts automatic mitigation for common errors.
 * @param {Error} error - The error that occurred
 * @param {Object} context - Error context (statusCode, etc.)
 * @returns {Object} Mitigation result { canRetry: boolean, action: string, handler?: Function }
 */
function attemptMitigation(error, context) {
  const { statusCode } = context;
  const categorized = categorizeError(error, statusCode);
  
  switch (categorized.category) {
    case 'NETWORK':
      return {
        canRetry: true,
        action: 'Retry with exponential backoff',
        handler: () => retryWithBackoff(confirmCommitToFlutterFlow),
      };
      
    case 'RATE_LIMIT':
      return {
        canRetry: true,
        action: 'Wait 60 seconds and retry',
        handler: () => delayRetry(60000),
      };
      
    case 'SERVER_ERROR':
      return {
        canRetry: true,
        action: 'Retry after server recovers (30s)',
        handler: () => delayRetry(30000),
      };
      
    case 'CONFLICT':
      return {
        canRetry: false,
        action: 'Manual resolution required - pull latest first',
        handler: null,
      };
      
    case 'AUTH':
      return {
        canRetry: false,
        action: 'Open API Keys settings to update credentials',
        handler: () => openApiKeysModal(),
      };
      
    case 'NOT_FOUND':
      return {
        canRetry: false,
        action: 'Open API Keys settings to verify Project ID',
        handler: () => openApiKeysModal(),
      };
      
    default:
      return {
        canRetry: true,
        action: 'Retry commit',
        handler: () => confirmCommitToFlutterFlow(),
      };
  }
}

/**
 * Retries a function with exponential backoff.
 * @param {Function} fn - Function to retry
 * @param {number} [maxRetries=3] - Maximum retry attempts
 * @param {number} [initialDelay=1000] - Initial delay in ms
 */
async function retryWithBackoff(fn, maxRetries = 3, initialDelay = 1000) {
  let delay = initialDelay;
  
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      console.log(`Retry attempt ${attempt}/${maxRetries}...`);
      return await fn();
    } catch (error) {
      if (attempt === maxRetries) {
        throw error;
      }
      
      console.log(`Attempt ${attempt} failed, waiting ${delay}ms...`);
      await sleep(delay);
      delay *= 2; // Exponential backoff
    }
  }
}

/**
 * Delays then retries the commit.
 * @param {number} delayMs - Delay in milliseconds
 */
async function delayRetry(delayMs) {
  console.log(`Waiting ${delayMs}ms before retry...`);
  await sleep(delayMs);
  return await confirmCommitToFlutterFlow();
}

/**
 * Utility sleep function.
 * @param {number} ms - Milliseconds to sleep
 * @returns {Promise<void>}
 */
function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

/**
 * Validates and sanitizes code before retry.
 * @param {string} code - Dart code
 * @returns {string} Sanitized code
 */
function sanitizeCodeForRetry(code) {
  let sanitized = code;
  
  // Remove common problematic patterns
  const problematicPatterns = [
    { pattern: /print\s*\([^)]*\);?\s*\n/g, replacement: '' }, // Remove print statements
    { pattern: /debugger;?\s*\n/g, replacement: '' }, // Remove debugger statements
  ];
  
  for (const { pattern, replacement } of problematicPatterns) {
    sanitized = sanitized.replace(pattern, replacement);
  }
  
  return sanitized;
}

/**
 * Attempts to fix common code issues automatically.
 * @param {string} code - Original code
 * @param {string[]} errors - Array of error messages
 * @returns {Object} Fix result { fixed: boolean, code: string, changes: string[] }
 */
function attemptAutoFix(code, errors) {
  let fixedCode = code;
  const changes = [];
  
  for (const error of errors) {
    // Fix missing width/height in Custom Widgets
    if (error.includes('width') || error.includes('height')) {
      if (!fixedCode.includes('double? width') && !fixedCode.includes('double? height')) {
        // Add width and height parameters to widget constructor
        fixedCode = fixedCode.replace(
          /class\s+(\w+)\s+extends\s+State/,
          'class $1 extends State'
        );
        changes.push('Widget may need width/height parameters');
      }
    }
    
    // Fix missing dispose
    if (error.includes('dispose') || error.includes('memory leak')) {
      if (!fixedCode.includes('void dispose()')) {
        changes.push('Widget may need dispose() method for controllers');
      }
    }
    
    // Fix import statements
    if (error.includes('import')) {
      // Remove all import lines (FlutterFlow manages imports)
      const originalLines = fixedCode.split('\n');
      fixedCode = originalLines
        .filter(line => !line.trim().startsWith('import '))
        .join('\n');
      if (originalLines.length !== fixedCode.split('\n').length) {
        changes.push('Removed import statements (FlutterFlow manages imports)');
      }
    }
  }
  
  return {
    fixed: changes.length > 0,
    code: fixedCode,
    changes,
  };
}
