/**
 * Error categories with mitigation steps.
 */
const ErrorCategories = {
  AUTH: {
    title: 'Authentication Failed',
    codes: [401],
    mitigation: [
      'Check that your FlutterFlow API Key is correct',
      'Verify the API Key has not expired in FlutterFlow Settings → API Access',
      'Generate a new API Key if needed and update it in API Keys settings',
    ],
  },
  FORBIDDEN: {
    title: 'Access Denied',
    codes: [403],
    mitigation: [
      'Verify you have permission to modify this project',
      'Check that you are a collaborator on the FlutterFlow project',
      'Contact the project owner to request access',
    ],
  },
  NOT_FOUND: {
    title: 'Project Not Found',
    codes: [404],
    mitigation: [
      'Verify your Project ID is correct',
      'Check that the project exists in your FlutterFlow account',
      'The Project ID is found in your FlutterFlow project URL',
    ],
  },
  CONFLICT: {
    title: 'Version Conflict',
    codes: [409],
    mitigation: [
      'The project has been modified elsewhere',
      'Pull the latest code from FlutterFlow first',
      'Reapply your changes and commit again',
    ],
  },
  VALIDATION: {
    title: 'Validation Failed',
    codes: [422],
    mitigation: [
      'Check that your code follows FlutterFlow guidelines',
      'Ensure no forbidden patterns (main(), runApp(), etc.)',
      'Verify all imports are FlutterFlow-compatible',
    ],
  },
  RATE_LIMIT: {
    title: 'Rate Limited',
    codes: [429],
    mitigation: [
      'Too many requests have been made',
      'Wait a few minutes before trying again',
      'Avoid rapid repeated commits',
    ],
  },
  SERVER_ERROR: {
    title: 'Server Error',
    codes: [500, 502, 503, 504],
    mitigation: [
      'FlutterFlow servers are experiencing issues',
      'Wait a few minutes and try again',
      'Check FlutterFlow status page for outages',
    ],
  },
  NETWORK: {
    title: 'Network Error',
    codes: ['NETWORK_ERROR', 'TIMEOUT'],
    mitigation: [
      'Check your internet connection',
      'Try again when you have a stable connection',
      'Disable VPN if you are using one',
    ],
  },
  VALIDATION_CLIENT: {
    title: 'Code Validation Failed',
    codes: ['VALIDATION_ERROR'],
    mitigation: [
      'Review the validation errors shown',
      'Fix any forbidden patterns in your code',
      'Ensure code follows FlutterFlow custom code guidelines',
    ],
  },
};

/**
 * Categorizes an error and returns mitigation steps.
 * @param {Error|string} error - Error object or message
 * @param {number} [statusCode] - HTTP status code if available
 * @returns {Object} Categorized error info
 */
function categorizeError(error, statusCode) {
  const errorMessage = error.message || error.toString();
  
  // Check by status code first
  for (const [category, info] of Object.entries(ErrorCategories)) {
    if (statusCode && info.codes.includes(statusCode)) {
      return { category, ...info };
    }
    // Check if any code matches error message
    for (const code of info.codes) {
      if (errorMessage.includes(code.toString())) {
        return { category, ...info };
      }
    }
  }
  
  // Check for common error patterns
  if (errorMessage.includes('network') || errorMessage.includes('fetch') || errorMessage.includes('internet')) {
    return { category: 'NETWORK', ...ErrorCategories.NETWORK };
  }
  if (errorMessage.includes('validation') || errorMessage.includes('invalid')) {
    return { category: 'VALIDATION_CLIENT', ...ErrorCategories.VALIDATION_CLIENT };
  }
  
  // Default to generic error
  return {
    category: 'UNKNOWN',
    title: 'Commit Failed',
    mitigation: [
      'An unexpected error occurred',
      'Try again in a few moments',
      'If the problem persists, check your credentials and try again',
    ],
  };
}

/**
 * Opens the commit error modal with categorized error info.
 * @param {Error} error - Error object
 * @param {Object} [context] - Additional context (statusCode, etc.)
 */
function openCommitErrorModal(error, context = {}) {
  const { statusCode } = context;
  const categorized = categorizeError(error, statusCode);
  
  // Set error info
  document.getElementById('error-title').textContent = categorized.title;
  document.getElementById('error-message').textContent = error.message || 'An error occurred';
  
  // Set error code if available
  const codeSection = document.getElementById('error-code-section');
  if (statusCode || categorized.category) {
    document.getElementById('error-code').textContent = statusCode 
      ? `HTTP ${statusCode} (${categorized.category})`
      : categorized.category;
    codeSection.classList.remove('hidden');
  } else {
    codeSection.classList.add('hidden');
  }
  
  // Set mitigation steps
  const mitigationList = document.getElementById('error-mitigation-list');
  mitigationList.innerHTML = categorized.mitigation
    .map(step => `<li class="flex items-start gap-2"><span class="text-orange-500 mt-1">•</span><span>${step}</span></li>`)
    .join('');
  
  // Set technical details
  document.getElementById('error-details-text').textContent = error.stack || error.toString();
  
  // Show modal
  const modal = document.getElementById('commit-error-modal');
  if (modal) {
    modal.classList.add('open');
  }
}

/**
 * Closes the commit error modal.
 * @param {Event} [event] - Optional click event
 */
function closeCommitErrorModal(event) {
  if (event && event.target !== event.currentTarget) return;
  const modal = document.getElementById('commit-error-modal');
  if (modal) {
    modal.classList.remove('open');
  }
}

/**
 * Toggles error details visibility.
 */
function toggleErrorDetails() {
  const content = document.getElementById('error-details-content');
  const chevron = document.getElementById('error-details-chevron');
  
  if (content.classList.contains('hidden')) {
    content.classList.remove('hidden');
    chevron.style.transform = 'rotate(90deg)';
  } else {
    content.classList.add('hidden');
    chevron.style.transform = 'rotate(0deg)';
  }
}
