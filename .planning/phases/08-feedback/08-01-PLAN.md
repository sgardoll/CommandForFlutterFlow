/**
 * Opens the commit success modal with commit details.
 * @param {Object} result - Commit result from executeCommit
 */
function openCommitSuccessModal(result) {
  // Populate fields
  document.getElementById('success-message').textContent = result.message || 'Code committed successfully!';
  document.getElementById('success-project-id').textContent = result.metadata?.projectId || 'Unknown';
  document.getElementById('success-file-name').textContent = result.metadata?.fileName || 'Unknown';
  document.getElementById('success-artifact-type').textContent = result.metadata?.artifactType || 'Unknown';
  document.getElementById('success-time').textContent = result.elapsedTime 
    ? `${(result.elapsedTime / 1000).toFixed(1)}s` 
    : 'Unknown';
  document.getElementById('success-size').textContent = result.metadata?.codeSize 
    ? `${(result.metadata.codeSize / 1024).toFixed(1)} KB` 
    : 'Unknown';
  
  // Populate warnings if any
  const warningsSection = document.getElementById('success-warnings-section');
  const warningsList = document.getElementById('success-warnings-list');
  if (result.warnings && result.warnings.length > 0) {
    warningsList.innerHTML = result.warnings
      .map(([file, warnings]) => {
        if (Array.isArray(warnings)) {
          return warnings.map(w => `<li>• ${file}: ${w.errorMessage || w}</li>`).join('');
        }
        return `<li>• ${file}</li>`;
      })
      .join('');
    warningsSection.classList.remove('hidden');
  } else {
    warningsSection.classList.add('hidden');
  }
  
  // Set FlutterFlow link
  const projectId = result.metadata?.projectId;
  const ffLink = document.getElementById('success-open-ff-link');
  if (projectId && ffLink) {
    ffLink.href = `https://app.flutterflow.io/project/${projectId}`;
  }
  
  // Show modal
  const modal = document.getElementById('commit-success-modal');
  if (modal) {
    modal.classList.add('open');
  }
}

/**
 * Closes the commit success modal.
 * @param {Event} [event] - Optional click event
 */
function closeCommitSuccessModal(event) {
  if (event && event.target !== event.currentTarget) return;
  const modal = document.getElementById('commit-success-modal');
  if (modal) {
    modal.classList.remove('open');
  }
}
